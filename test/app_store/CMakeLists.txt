cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(app_store_behavior_tests LANGUAGES CXX)
  enable_testing()
endif()
set(ARDUINOJSON_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../.pio/libdeps/simulator/ArduinoJson/src" CACHE PATH "Path to existing ArduinoJson/src (no download)")
if(NOT EXISTS "${ARDUINOJSON_INCLUDE_DIR}/ArduinoJson.h")
  message(FATAL_ERROR "Set ARDUINOJSON_INCLUDE_DIR to an existing ArduinoJson/src directory")
endif()
add_executable(app_store_test app_store_test.cpp)
target_compile_features(app_store_test PRIVATE cxx_std_17)
target_include_directories(app_store_test PRIVATE stubs ../../src "${ARDUINOJSON_INCLUDE_DIR}")
target_compile_options(app_store_test PRIVATE -Wall -Wextra -Wno-unused-variable)
option(APP_STORE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(APP_STORE_SANITIZERS)
  target_compile_options(app_store_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(app_store_test PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME app_store_behavior COMMAND app_store_test)
